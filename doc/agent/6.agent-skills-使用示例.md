### 加载技能

```typescript
import {
  loadWorkspaceSkillEntries,
  buildWorkspaceSkillSnapshot,
} from './src/agents/skills/index.js';

async function basicExample() {
  const workspaceDir = process.cwd();

  // 加载所有技能
  const entries = loadWorkspaceSkillEntries(workspaceDir);
  console.log(`✓ Loaded ${entries.length} skills`);

  // 构建技能快照
  const snapshot = buildWorkspaceSkillSnapshot(workspaceDir);
  console.log(`✓ Built skill snapshot`);
  console.log(`✓ Skills prompt length: ${snapshot.prompt.length} chars`);
  console.log(`✓ Included ${snapshot.skills.length} skills`);
}
```

### 列出技能

```typescript
import { loadWorkspaceSkillEntries } from './src/agents/skills/index.js';

function listSkills() {
  const workspaceDir = process.cwd();
  const entries = loadWorkspaceSkillEntries(workspaceDir);

  console.log('Available Skills:');
  console.log('='.repeat(50));

  for (const entry of entries) {
    const emoji = entry.metadata?.emoji || '🔧';
    const status = entry.invocation?.userInvocable ? '✓' : '✗';
    const description = entry.skill.description.slice(0, 60);

    console.log(`${emoji} ${entry.skill.name}`);
    console.log(`   ${description}${description.length >= 60 ? '...' : ''}`);
    console.log(`   User invocable: ${status}`);
    console.log(`   Source: ${entry.skill.source}`);
    console.log();
  }
}
```

## 技能过滤

### 使用名称过滤器

```typescript
import {
  loadWorkspaceSkillEntries,
  buildWorkspaceSkillSnapshot,
  SkillMatcher,
} from './src/agents/skills/index.js';

function filterByName() {
  const workspaceDir = process.cwd();
  const entries = loadWorkspaceSkillEntries(workspaceDir);

  // 创建名称过滤器
  const matcher = new SkillMatcher(['github', 'git', 'code']);

  // 过滤技能
  const filtered = entries.filter(e => matcher.matches(e.skill.name));

  console.log(`Filtered ${entries.length} skills to ${filtered.length}`);

  // 构建快照
  const snapshot = buildWorkspaceSkillSnapshot(workspaceDir, {
    entries: filtered,
  });

  console.log('Skills prompt:', snapshot.prompt);
}
```

### 使用标签过滤器

```typescript
import {
  loadWorkspaceSkillEntries,
  SkillTagMatcher,
} from './src/agents/skills/index.js';

function filterByTags() {
  const workspaceDir = process.cwd();
  const entries = loadWorkspaceSkillEntries(workspaceDir);

  // 创建标签过滤器
  const tagMatcher = new SkillTagMatcher({
    require: ['git', 'development'],
    exclude: ['experimental'],
  });

  // 过滤技能
  const filtered = entries.filter(e => {
    const tags = SkillTagMatcher.extractTags(e.frontmatter);
    return tagMatcher.matches(tags);
  });

  console.log(`Skills with git and development tags: ${filtered.length}`);
}
```

## 配置管理

### 配置技能启用状态

```typescript
import {
  loadWorkspaceSkillEntries,
  buildWorkspaceSkillSnapshot,
} from './src/agents/skills/index.js';

function configureSkills() {
  const workspaceDir = process.cwd();

  // 定义配置
  const config = {
    skills: {
      entries: {
        github: {
          enabled: true,
          apiKey: process.env.GITHUB_TOKEN,
        },
        git: {
          enabled: false, // 禁用 git 技能
        },
      },
      limits: {
        maxSkillsInPrompt: 50, // 限制提示词中的技能数量
        maxSkillsPromptChars: 15000, // 限制提示词字符数
      },
    },
  };

  // 加载技能 (会应用配置)
  const entries = loadWorkspaceSkillEntries(workspaceDir, config);

  // 构建快照
  const snapshot = buildWorkspaceSkillSnapshot(workspaceDir, {
    config,
  });

  console.log('Skills in prompt:', snapshot.skills.map(s => s.name));
}
```

## 集成到 Agent

### 基本集成

```typescript
import { Agent } from './src/agents/agent.js';
import {
  buildWorkspaceSkillSnapshot,
  type SkillSnapshot,
} from './src/agents/skills/index.js';

class SkillsEnabledAgent extends Agent {
  private skillsSnapshot?: SkillSnapshot;

  constructor(config: any) {
    super(config);
  }

  async initializeSkills(workspaceDir: string) {
    // 构建技能快照
    this.skillsSnapshot = buildWorkspaceSkillSnapshot(workspaceDir, {
      config: this.config.skills,
    });

    console.log(`[Agent] Initialized ${this.skillsSnapshot.skills.length} skills`);
  }

  protected buildContext(): string {
    // 获取基础上下文
    const baseContext = super.buildContext();

    // 添加技能上下文
    const skillsContext = this.skillsSnapshot?.prompt || '';

    return `${baseContext}\n\n${skillsContext}`;
  }
}

// 使用
async function useSkillsAgent() {
  const agent = new SkillsEnabledAgent({
    id: 'skills-agent',
    model: 'openai/gpt-4',
  });

  await agent.initializeSkills(process.cwd());
  await agent.connect();
}
```

### 高级集成

```typescript
import { Agent } from './src/agents/agent.js';
import {
  loadWorkspaceSkillEntries,
  buildWorkspaceSkillSnapshot,
  buildWorkspaceSkillCommandSpecs,
  SkillMatcher,
} from './src/agents/skills/index.js';

class AdvancedSkillsAgent extends Agent {
  private skillSnapshot?: import('./src/agents/skills/types.js').SkillSnapshot;
  private skillCommands: Map<string, any> = new Map();

  async initializeSkills(workspaceDir: string, config?: any) {
    // 加载技能
    const entries = loadWorkspaceSkillEntries(workspaceDir, config);

    // 过滤技能 (只启用的)
    const matcher = new SkillMatcher(['github', 'git', 'code', 'devops']);
    const filtered = entries.filter(e => matcher.matches(e.skill.name));

    // 构建快照
    this.skillSnapshot = buildWorkspaceSkillSnapshot(workspaceDir, {
      entries: filtered,
      config,
    });

    // 构建命令规范
    const commands = buildWorkspaceSkillCommandSpecs(workspaceDir, {
      entries: filtered,
    });

    // 注册技能命令
    for (const cmd of commands) {
      this.registerSkillCommand(cmd);
    }

    console.log(`[Agent] Initialized ${filtered.length} skills, ${commands.length} commands`);
  }

  private registerSkillCommand(cmd: any) {
    this.registerTool({
      name: cmd.name,
      description: cmd.description,
      handler: async (args) => {
        console.log(`[Agent] Executing skill command: ${cmd.skillName}`);
        // 技能命令处理逻辑
        return { success: true, result: 'Skill executed' };
      },
    });
  }

  protected buildContext(): string {
    const baseContext = super.buildContext();
    const skillsContext = this.skillSnapshot?.prompt || '';
    return `${baseContext}\n\n${skillsContext}`;
  }
}
```

### 创建 GitHub 技能

```bash
mkdir -p skills/github-helper/scripts

cat > skills/github-helper/SKILL.md << 'EOF'
---
name: github-helper
description: GitHub 操作辅助
emoji: 🐙
primaryEnv: GITHUB_TOKEN
requires:
  bins: [git]
  env: [GITHUB_TOKEN]
---

# GitHub Helper

GitHub 操作辅助技能。

## 功能

- 创建和管理 Issues
- 创建和管理 Pull Requests
- 查询仓库信息

## 使用示例

用户：在 user/repo 创建一个 Issue，标题是 "Bug"，内容是 "详细描述"
EOF

cat > skills/github-helper/scripts/create-issue.sh << 'EOF'
#!/usr/bin/env bash
set -e

REPO="$1"
TITLE="$2"
BODY="${3:-}"

if [ -z "$GITHUB_TOKEN" ]; then
  echo "Error: GITHUB_TOKEN not set"
  exit 1
fi

RESPONSE=$(curl -s -X POST "https://api.github.com/repos/${REPO}/issues" \
  -H "Authorization: token $GITHUB_TOKEN" \
  -H "Accept: application/vnd.github.v3+json" \
  -d "{\"title\":\"${TITLE}\",\"body\":\"${BODY}\"}")

echo "$RESPONSE"
EOF

chmod +x skills/github-helper/scripts/create-issue.sh

echo "✓ Created GitHub helper skill"
```

## 完整示例

### 技能管理 CLI

```typescript
#!/usr/bin/env node

import {
  loadWorkspaceSkillEntries,
  buildWorkspaceSkillSnapshot,
  buildWorkspaceSkillCommandSpecs,
} from './src/agents/skills/index.js';

const command = process.argv[2];
const workspaceDir = process.cwd();

async function main() {
  switch (command) {
    case 'list': {
      const entries = loadWorkspaceSkillEntries(workspaceDir);
      console.log('Available Skills:');
      for (const entry of entries) {
        console.log(`  - ${entry.skill.name}: ${entry.skill.description}`);
      }
      break;
    }

    case 'snapshot': {
      const snapshot = buildWorkspaceSkillSnapshot(workspaceDir);
      console.log('Skills Prompt:');
      console.log(snapshot.prompt);
      break;
    }

    case 'commands': {
      const commands = buildWorkspaceSkillCommandSpecs(workspaceDir);
      console.log('Available Commands:');
      for (const cmd of commands) {
        console.log(`  /${cmd.name} - ${cmd.description}`);
      }
      break;
    }

    default:
      console.log('Usage: node skills-cli.js <command>');
      console.log('Commands: list, snapshot, commands');
  }
}

main().catch(console.error);
```

## 注意事项

1. **环境变量** - 确保技能需要的环境变量已设置
2. **路径解析** - 技能路径会被压缩以节省 tokens
3. **并发安全** - 技能加载是同步的，注意并发问题
4. **错误处理** - 始终处理技能加载和执行错误
5. **性能考虑** - 合理设置技能数量和字符数限制
